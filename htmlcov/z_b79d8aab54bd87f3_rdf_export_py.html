1"""SQLite triple store -> RDF/Turtle export (Tier B, foundation layer). 

2 

3Turns Tier A's live `TripleStore` (graph_guard/store.py) into a formal RDF graph that 

4later Tier B tasks validate (SHACL), reason over (owlrl) and query (SPARQL). Tier A stays 

5the live retrieval engine; this module is a read-only, pure export/interop layer -- it 

6never mutates the store. 

7 

8Namespaces (D2): 

9 kl: https://jott2121.github.io/graph-guard/ns# (this project's vocabulary) 

10 schema: https://schema.org/ 

11 skos: http://www.w3.org/2004/02/skos/core# 

12 plus rdflib's built-in rdf, rdfs, owl, xsd. 

13 

14IRI minting (D2): 

15 - node IRI: kl:n/<slug> where <slug> is a URL-safe, deterministic transform of the 

16 node id (non `[A-Za-z0-9_-]` bytes percent-encoded via urllib.parse.quote, 

17 safe="", so the mapping is stable and reversible via unquote). 

18 - predicate IRI: kl:<predicate> (predicates are already a closed, safe vocabulary -- 

19 see graph_guard/schema.py PREDICATES). 

20 - type IRI: kl:<Type> from the node's `type` field (e.g. kl:Project). 

21 - node name: emitted as `schema:name` (not rdfs:label) -- Task 3's SHACL required-name 

22 check is written against schema:name. 

23 

24Provenance/confidence (D1 -- standard RDF reification, not RDF-star): 

25 This rdflib build has no ttlstar plugin, and the downstream SHACL/owlrl toolchain is only 

26 reliable over plain RDF. So each edge produces BOTH: 

27 1. the asserted triple <src IRI> <predicate IRI> <dst IRI> (what SPARQL property 

28 paths and owlrl reason over), and 

29 2. a reification block on a deterministically minted statement IRI `kl:stmt/<n>` (n = 

30 the edge's position in TripleStore.all_edges(), stable because SQLite returns rows in 

31 insertion/PK order for a given store): 

32 kl:stmt/<n> a rdf:Statement ; 

33 rdf:subject <src IRI> ; 

34 rdf:predicate <predicate IRI> ; 

35 rdf:object <dst IRI> ; 

36 kl:confidence "<confidence>"^^xsd:decimal ; 

37 kl:provenance "<note_path>" ; # omitted if null 

38 kl:span "<span>" . # omitted if null 

39 A minted IRI (rather than a random/blank bnode) keeps the export deterministic so tests 

40 can assert on it directly. 

41""" 

42from __future__ import annotations 

43 

44from urllib.parse import quote 

45 

46from rdflib import RDF, Graph, Literal, Namespace, URIRef 

47from rdflib.namespace import XSD 

48 

49KL = Namespace("https://jott2121.github.io/graph-guard/ns#") 

50SCHEMA = Namespace("https://schema.org/") 

51SKOS = Namespace("http://www.w3.org/2004/02/skos/core#") 

52 

53_SAFE = "-_" # everything else outside [A-Za-z0-9] gets percent-encoded 

54 

55 

56def node_iri(node_id: str) -> URIRef: 

57 """Deterministic node IRI: kl:n/<percent-encoded id>. Same id -> same IRI; any string 

58 (spaces, slashes, unicode) yields a valid, parseable IRI.""" 

59 return KL[f"n/{quote(str(node_id), safe=_SAFE)}"] 

60 

61 

62def predicate_iri(predicate: str) -> URIRef: 

63 return KL[predicate] 

64 

65 

66def type_iri(type_name: str) -> URIRef: 

67 return KL[type_name] 

68 

69 

70def stmt_iri(index: int) -> URIRef: 

71 return KL[f"stmt/{index}"] 

72 

73 

74def _bind_namespaces(g: Graph) -> None: 

75 g.bind("kl", KL) 

76 g.bind("schema", SCHEMA) 

77 g.bind("skos", SKOS) 

78 

79 

80def store_to_graph(store) -> Graph: 

81 """Build an in-memory rdflib.Graph from a TripleStore. Pure: touches only the store's 

82 read methods, never writes to disk.""" 

83 g = Graph() 

84 _bind_namespaces(g) 

85 

86 for node in store.all_nodes(): 

87 n_iri = node_iri(node["id"]) 

88 g.add((n_iri, RDF.type, type_iri(node["type"]))) 

89 g.add((n_iri, SCHEMA.name, Literal(node["name"]))) 

90 

91 for i, edge in enumerate(store.all_edges()): 

92 s_iri = node_iri(edge["src"]) 

93 p_iri = predicate_iri(edge["predicate"]) 

94 o_iri = node_iri(edge["dst"]) 

95 g.add((s_iri, p_iri, o_iri)) 

96 

97 stmt = stmt_iri(i) 

98 g.add((stmt, RDF.type, RDF.Statement)) 

99 g.add((stmt, RDF.subject, s_iri)) 

100 g.add((stmt, RDF.predicate, p_iri)) 

101 g.add((stmt, RDF.object, o_iri)) 

102 g.add((stmt, KL.confidence, Literal(edge["confidence"], datatype=XSD.decimal))) 

103 if edge.get("note_path"): 

104 g.add((stmt, KL.provenance, Literal(edge["note_path"]))) 

105 if edge.get("span"): 

106 g.add((stmt, KL.span, Literal(edge["span"]))) 

107 

108 return g 

109 

110 

111def export_turtle(store, path=None) -> str: 

112 """Serialize the store to Turtle. Writes to `path` if given; always returns the string.""" 

113 ttl = store_to_graph(store).serialize(format="turtle") 

114 if path is not None: 

115 with open(path, "w", encoding="utf-8") as f: 

116 f.write(ttl) 

117 return ttl
