1"""owlrl-materialized RDF graph -> note-id adjacency bridge (Gate D, Task 1). 

2 

3Gate D's "reasoned" retrieval arm (A3) runs Personalized PageRank (`graph_guard.ppr`) over 

4the OWL-entailed graph instead of the raw KG, to see whether owlrl's entailments (the 

5`kl:supersedesTransitively` closure, `skos:broader` rollups, subclass typing, ...) lift 

6multi-hop retrieval beyond what the raw `TripleStore.adjacency()` graph arm (A2) already 

7gives PageRank. But `personalized_pagerank` only understands a note-id adjacency dict, and 

8`reasoning.materialize` only understands RDF/IRI space -- this module is the bridge: it 

9starts from the store's asserted adjacency, then folds in the entailed graph, returning a 

10`{note_id: {neighbor_id: weight}}` dict shaped exactly like `TripleStore.adjacency()`, so it 

11drops straight into `ppr.personalized_pagerank`. 

12 

13**A3 = A2 + only-the-edges-reasoning-reveals.** So the arms differ ONLY by what reasoning 

14adds, never by re-weighting: 

15 - Start from a deep copy of `store.adjacency()` -- this carries every ASSERTED edge at its 

16 real confidence weight (the graph arm A2). Deep-copied so the store's structure is never 

17 mutated. 

18 - Then walk the materialized graph and add an entailed note<->note edge ONLY for a pair 

19 that is NOT already connected in that starting adjacency (`o_id in adj[s_id]` -> skip). 

20 That leaves every asserted edge at its original confidence and adds only genuinely-new 

21 reachability -- e.g. the transitive A->C/A->D supersedes closure -- at a uniform weight 

22 of 1.0. Uniform because entailed triples carry no confidence of their own (confidence 

23 lives only in `rdf_export`'s reification blocks, keyed to a specific ASSERTED triple, not 

24 to whatever an owlrl rule derives from it). 

25 

26Projection rule for what counts as an entailed edge: a materialized triple `(s, p, o)` 

27becomes a candidate note<->note edge only when BOTH `s` and `o` resolve (`fuseki._iri_to_id`) 

28to real note ids already in the store, AND `p` is a relational `kl:`/`skos:` predicate -- 

29never `rdf:type`, `schema:name`, or any triple in the reification vocabulary 

30(`rdf:Statement`/`rdf:subject`/`rdf:predicate`/`rdf:object`/`kl:confidence`/`kl:provenance`/ 

31`kl:span`) that `rdf_export` emits alongside every asserted edge. That keeps entailed 

32*typing* (e.g. a `Project` note's entailed `rdf:type schema:Project`) and provenance 

33bookkeeping from ever leaking into the graph as a spurious note<->note edge. 

34 

35`type_cohort=True` is an off-by-default experiment toggle: it additionally connects every 

36pair of notes that share an entailed **kl: entity class** (grouping note ids by their 

37`rdf:type kl:<Type>`). It deliberately ignores the universal types owlrl materializes on 

38every individual (`owl:Thing`, `rdfs:Resource`) and the external super-classes 

39(`schema:*`/`skos:*`) -- grouping by those would link every note to every other note (the 

40`owl:Thing` cohort) and make the toggle meaningless. Even restricted to kl: types this is a 

41much noisier signal than the relational edges above (two unrelated `Project` notes get 

42linked just for sharing a type), so it's kept opt-in for Gate D to measure separately rather 

43than baked into the default reasoned arm. 

44""" 

45from __future__ import annotations 

46 

47import copy 

48 

49from rdflib import RDF 

50 

51from graph_guard import rdf_export, reasoning 

52from graph_guard.fuseki import _iri_to_id 

53from graph_guard.rdf_export import KL, SCHEMA, SKOS 

54 

55# Predicates that must never become adjacency edges: RDF typing, the node's display 

56# name, and every predicate in the reification vocabulary rdf_export.store_to_graph 

57# emits per asserted edge (a `kl:stmt/<n> a rdf:Statement` block carrying 

58# rdf:subject/predicate/object + kl:confidence/provenance/span). None of these encode 

59# a note<->note relation, even after owlrl materialization. 

60_EXCLUDE_PREDICATES = frozenset({ 

61 RDF.type, 

62 SCHEMA.name, 

63 RDF.subject, 

64 RDF.predicate, 

65 RDF.object, 

66 KL.confidence, 

67 KL.provenance, 

68 KL.span, 

69}) 

70 

71_RELATIONAL_NAMESPACES = (str(KL), str(SKOS)) 

72 

73_ENTAILED_EDGE_WEIGHT = 1.0 

74 

75 

76def _add_edge(adj: dict[str, dict[str, float]], a: str, b: str, weight: float) -> None: 

77 adj.setdefault(a, {}) 

78 adj.setdefault(b, {}) 

79 adj[a][b] = adj[a].get(b, 0.0) + weight 

80 adj[b][a] = adj[b].get(a, 0.0) + weight 

81 

82 

83def reasoned_adjacency(store, *, type_cohort: bool = False) -> dict[str, dict[str, float]]: 

84 """Return A3: the asserted graph-arm adjacency (`store.adjacency()`, at its real 

85 confidence weights) PLUS only the genuinely-new note<->note edges OWL reasoning reveals 

86 (weight 1.0). Same `{node_id: {neighbor_id: weight}}` shape `TripleStore.adjacency()` 

87 returns, so it's a drop-in for `ppr.personalized_pagerank`. 

88 

89 Asserted edges keep their confidence untouched; an entailed edge is added only for a pair 

90 not already connected. See the module docstring for exactly what counts as an entailed 

91 edge vs. what's excluded (typing, reification), and why `type_cohort` is opt-in. 

92 """ 

93 g = reasoning.materialize(rdf_export.store_to_graph(store)) 

94 note_ids = {n["id"] for n in store.all_nodes()} 

95 

96 # A2, at real confidence weights. Deep copy so the store's dict is never mutated; only 

97 # genuinely-new entailed edges are folded in on top. 

98 adj: dict[str, dict[str, float]] = copy.deepcopy(store.adjacency()) 

99 

100 for s, p, o in g: 

101 if p in _EXCLUDE_PREDICATES: 

102 continue 

103 if not str(p).startswith(_RELATIONAL_NAMESPACES): 

104 continue 

105 s_id = _iri_to_id(str(s)) 

106 if s_id not in note_ids: 

107 continue 

108 o_id = _iri_to_id(str(o)) 

109 if o_id not in note_ids: 

110 continue 

111 if o_id in adj.get(s_id, {}): 

112 continue # already connected in A2 (or a prior entailed add) -> preserve its weight 

113 _add_edge(adj, s_id, o_id, _ENTAILED_EDGE_WEIGHT) 

114 

115 if type_cohort: 

116 cohorts: dict[str, list[str]] = {} 

117 for s, _p, o in g.triples((None, RDF.type, None)): 

118 if not str(o).startswith(str(KL)): 

119 continue # kl: entity classes only -- skip owl:Thing/rdfs:Resource/schema:/skos: 

120 s_id = _iri_to_id(str(s)) 

121 if s_id not in note_ids: 

122 continue 

123 cohorts.setdefault(str(o), []).append(s_id) 

124 for members in cohorts.values(): 

125 for i in range(len(members)): 

126 for j in range(i + 1, len(members)): 

127 _add_edge(adj, members[i], members[j], _ENTAILED_EDGE_WEIGHT) 

128 

129 return adj
